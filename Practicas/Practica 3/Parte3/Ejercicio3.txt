EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H

N_F10 EQU 10

ORG 40
DIR_F10 DW RUT_F10

ORG 1000H
SELECCIONADA DB 30H
FIN DB ?

ORG 2000H
;; CONFIGURO EL IMR PARA QUE ACEPTE LA INTERRUPCIÓN DE F10
CLI
  MOV AL, 11111110B
  OUT IMR, AL
  MOV AL, N_F10
  OUT INT0, AL
STI
REINICIAR:
  MOV DL, 0
  LOOP:
  INC DL
  CMP DL, 9
  JZ REINICIAR
  CMP FIN, 1
  JNZ LOOP
  HLT
  
ORG 3000H
RUT_F10:
  PUSH AX
  PUSH BX
  MOV BX, OFFSET SELECCIONADA
  ADD [BX],DL
  MOV AL, 1
  MOV FIN, 1
  INT 7
  MOV AL, 20H
  OUT EOI, AL
  POP BX 
  POP AX
IRET

END
