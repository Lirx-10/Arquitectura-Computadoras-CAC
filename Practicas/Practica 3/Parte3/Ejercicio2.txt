EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H

PB EQU 31H
CB EQU 33H

N_F10 EQU 10

ORG 40
DIR_F10 DW RUT_F10

ORG 2000H
;; CONFIGURO LA SALIDA PARA PRENDER LAS LUCES
  MOV AL, 0
  OUT CB, AL
  MOV AL, 0FFH
  OUT PB, AL

CLI
;; CONFIGURO EL IMR PARA QUE ACEPTE LA INTERRUPCIÓN DE F10
  MOV AL, 11111110B
  OUT IMR, AL
  MOV AL, N_F10
  OUT INT0, AL
STI
LOOP:
  JMP LOOP
  
ORG 3000H
RUT_F10:
  PUSH AX
  IN AL, PB
  CMP AL, 0FFH
  JNZ PRENDER
  APAGAR:
  MOV AL, 0
  OUT PB, AL
  JMP FIN
  PRENDER:
  MOV AL, 0FFH
  OUT PB, AL
  FIN:
  MOV AL, 20H
  OUT EOI, AL
  POP AX
IRET

END
