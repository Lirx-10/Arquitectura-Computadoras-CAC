POR CADA VEZ QUE APRETAMOS F10 SE INCREMENTA EL CONTADOR EN DX

EOI EQU 20H
IMR EQU 21H
INT0 EQU 24H
N_F10 EQU 10

ORG 40
DIR_F10 DW RUT_F10

ORG 2000H
CLI
;; CONFIGURO EL IMR PARA QUE TENGA HABILITADA LA INTERRUPCION F10
MOV AL, 11111110B
OUT IMR, AL
;;
MOV AL, N_F10
OUT INT0, AL
MOV DX, 0
STI
LAZO:
  JMP LAZO
  
ORG 3000H
RUT_F10:
  PUSH AX
  INC DX
  MOV AL, 20H
  OUT EOI, AL
  POP AX
IRET

END
