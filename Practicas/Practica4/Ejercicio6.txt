Escribir un programa que lea tres números enteros A, B y C de la memoria de datos y determine 
cuántos de ellos son iguales entre sí (0, 2 o 3). El resultado debe quedar almacenado en la 
dirección de memoria D.

.data
A: .word 1
B: .word 1
C: .word 1
D: .word 0

.code
ld r1, A(r0)
ld r2, B(r0)
ld r3, C(r0)

beq r1, r2, siguiente
beq r1, r3, solo2
beq r2, r3, solo2
ninguna:
    dadd r4,r0,r0
    j fin
siguiente: 
    beq r1, r3, todas
solo2:
    daddi r4, r0, 2
    j fin
todas:
    daddi r4, r0, 3 
    j fin
fin:
sd r4, D(r0)

halt
